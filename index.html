from urllib.parse import quote

def https_host(src: str) -> str:
    """
    Example:
        >>> html_source = "<script>alert('hi!')</script>"
        >>> https_host(html_source)
        'https://reflect.ahh.bet/#%3Cscript%3Ealert%28%27hi%21%27%29%3C%2Fscript%3E'
    """
    html = quote(src)
    return f"https://reflect.ahh.bet/#{html}"

JavaScript Usage

// Click anywhere to copy!
function createReflectLink(htmlString) {
    const encodedHtml = encodeURIComponent(htmlString);
    return `https://reflect.ahh.bet/#${encodedHtml}`;
}
